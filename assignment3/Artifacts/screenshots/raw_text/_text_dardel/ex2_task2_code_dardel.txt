    int *frontier = (int *)malloc((size_t)n * sizeof(int));
    int *next_frontier = (int *)malloc((size_t)n * sizeof(int));
    if (!frontier || !next_frontier) {
        die("Failed to allocate frontier buffers");
    }

    for (int i = 0; i < n; i++) {
        dist[i] = -1;
    }
    int frontier_size = 1;
    frontier[0] = source;
    dist[source] = 0;
    int level = 0;

    while (frontier_size > 0) {
        int next_size = 0;

#pragma omp parallel
        {
#pragma omp single nowait
            {
                int chunk = 64;
                for (int start = 0; start < frontier_size; start += chunk) {
                    int end = start + chunk;
                    if (end > frontier_size) {
                        end = frontier_size;
                    }

#pragma omp task firstprivate(start, end, level) shared(frontier, next_frontier, dist, next_size)
                    {
                        int cap = n > 1024 ? 1024 : n;
                        int *local_next = (int *)malloc((size_t)cap * sizeof(int));
                        int local_count = 0;
                        if (!local_next) {
                            die("Failed to allocate local task buffer");
                        }

                        for (int i = start; i < end; i++) {
                            int u = frontier[i];
                            for (int e = g->row_ptr[u]; e < g->row_ptr[u + 1]; e++) {
                                int v = g->col_idx[e];
                                if (__sync_bool_compare_and_swap(&dist[v], -1, level + 1)) {
                                    if (local_count == cap) {
                                        cap *= 2;
                                        int *tmp = (int *)realloc(local_next, (size_t)cap * sizeof(int));
                                        if (!tmp) {
                                            free(local_next);
                                            die("Failed to realloc local task buffer");
                                        }
                                        local_next = tmp;
                                    }
                                    local_next[local_count++] = v;
                                }
                            }
                        }

                        int base;
#pragma omp atomic capture
                        {
                            base = next_size;
                            next_size += local_count;
                        }
                        memcpy(next_frontier + base, local_next, (size_t)local_count * sizeof(int));
                        free(local_next);
                    }
                }
            }
        }

        int *tmp = frontier;
        frontier = next_frontier;
        next_frontier = tmp;
        frontier_size = next_size;
        level++;
    }

    free(frontier);
    free(next_frontier);
}

static int verify_result(const int *a, const int *b, int n) {
    for (int i = 0; i < n; i++) {
        if (a[i] != b[i]) {
            return 0;
        }
    }
    return 1;
}

int main(int argc, char **argv) {
    int n = 50000;
    int degree = 8;
    if (argc > 1) {
        n = atoi(argv[1]);
    }
    if (argc > 2) {
        degree = atoi(argv[2]);
    }
    if (n <= 1 || degree <= 0) {
        die("Usage: ./bfs_ex2_task2_task [num_vertices] [degree]");
    }

    GraphCSR g = generate_random_graph(n, degree, 2027U);
    int *dist_serial = (int *)malloc((size_t)n * sizeof(int));
    int *dist_task = (int *)malloc((size_t)n * sizeof(int));
    if (!dist_serial || !dist_task) {
        die("Failed to allocate distance arrays");
    }

    double t0 = omp_get_wtime();
    bfs_serial(&g, 0, dist_serial);
    double t1 = omp_get_wtime();

    double t2 = omp_get_wtime();
    bfs_omp_task(&g, 0, dist_task);
    double t3 = omp_get_wtime();

    int ok = verify_result(dist_serial, dist_task, n);
    int visited = 0;
    for (int i = 0; i < n; i++) {
        if (dist_task[i] >= 0) {
            visited++;
        }
    }

    printf("Graph: vertices=%d edges=%d degree=%d\n", n, g.m, degree);
    printf("OMP max threads: %d\n", omp_get_max_threads());
    printf("Serial time:   %.6f s\n", t1 - t0);
    printf("OMP time:      %.6f s\n", t3 - t2);
    printf("Speedup:       %.3f x\n", (t1 - t0) / (t3 - t2));
    printf("Visited:       %d\n", visited);
    printf("Verification:  %s\n", ok ? "PASS" : "FAIL");

    free(dist_serial);
    free(dist_task);
    free_graph(&g);
    return ok ? 0 : 2;
}
